cmake_minimum_required(VERSION 3.14)
project(folio_clipboard_test LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "Run this test with the Windows toolchain")
endif()
add_executable(windows_rich_clipboard_test ../windows_rich_clipboard_test.cpp)
target_compile_features(windows_rich_clipboard_test PRIVATE cxx_std_20)
target_compile_definitions(windows_rich_clipboard_test PRIVATE NOMINMAX)
target_link_libraries(windows_rich_clipboard_test PRIVATE user32)
enable_testing()
add_test(NAME windows_rich_clipboard COMMAND windows_rich_clipboard_test)
